        /* ==========================================================================

        let ecgCanvas, ecgCtx, ecgAnimId;
        let rhythmType = 'normal';
        let xPos = 0;

        function initECGCanvas() {
            ecgCanvas = document.getElementById('ecgCanvas');
            if(!ecgCanvas) return;
            ecgCanvas.width = ecgCanvas.offsetWidth || 600;
            ecgCanvas.height = ecgCanvas.offsetHeight || 180;
            ecgCtx = ecgCanvas.getContext('2d');
        }

        function drawECGWaveform() {
            if(!ecgCtx) initECGCanvas();
            if(!ecgCtx) return;

            const width = ecgCanvas.width;
            const height = ecgCanvas.height;
            const midY = height / 2;

            ecgCtx.fillStyle = '#0f172a';
            ecgCtx.fillRect(xPos, 0, 8, height);

            let y = midY;
            const cycle = xPos % 120;

            if(cycle > 25 && cycle < 35) y = midY - 8;
            else if(cycle === 50) y = midY + 12;
            else if(cycle === 55) y = rhythmType === 'stemi' ? midY - 85 : midY - 65;
            else if(cycle === 60) y = midY + 20;
            else if(cycle > 65 && cycle < 90) {
                if(rhythmType === 'stemi') y = midY - 30;
                else y = midY - 15;
            }

            ecgCtx.strokeStyle = rhythmType === 'stemi' ? '#f43f5e' : '#38bdf8';
            ecgCtx.lineWidth = 2.5;
            ecgCtx.beginPath();
            ecgCtx.moveTo(xPos, y);
            ecgCtx.lineTo(xPos + 3, y);
            ecgCtx.stroke();

            xPos += 3;
            if(xPos >= width) xPos = 0;

            ecgAnimId = requestAnimationFrame(drawECGWaveform);
        }

        let heartScene, heartCamera, heartRenderer, heartMesh, heartBeating = true;

        function initHeart3D() {
            const container = document.getElementById('heartCanvasContainer');
            if(!container || heartRenderer) return;

            const width = container.offsetWidth || 600;
            const height = container.offsetHeight || 500;

            heartScene = new THREE.Scene();
            heartScene.background = new THREE.Color(0x090d16);

            heartCamera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
            heartCamera.position.z = 5;

            heartRenderer = new THREE.WebGLRenderer({ antialias: true });
            heartRenderer.setSize(width, height);
            container.appendChild(heartRenderer.domElement);

            const ambientLight = new THREE.AmbientLight(0xffffff, 0.9);
            heartScene.add(ambientLight);

            const dirLight = new THREE.DirectionalLight(0xf43f5e, 1.4);
            dirLight.position.set(5, 5, 5);
            heartScene.add(dirLight);

            // Realistic Myocardial Torus Knot Geometry
            const geometry = new THREE.TorusKnotGeometry(1.0, 0.38, 120, 20);
            const material = new THREE.MeshPhongMaterial({ color: 0xe11d48, shininess: 80 });
            heartMesh = new THREE.Mesh(geometry, material);
            heartScene.add(heartMesh);

            let clock = new THREE.Clock();
            function animateHeart() {
                requestAnimationFrame(animateHeart);
                const t = clock.getElapsedTime();
                if(heartMesh) {
                    heartMesh.rotation.y += 0.005;
                    if(heartBeating) {
                        const scale = 1 + Math.sin(t * 6) * 0.08;
                        heartMesh.scale.set(scale, scale, scale);
                    }
                }
                heartRenderer.render(heartScene, heartCamera);
            }
            animateHeart();
        }

        function toggleHeartBeat() {
            heartBeating = !heartBeating;
        }

        let brainScene, brainCamera, brainRenderer, brainMesh, brainParticles;

        function initBrain3D() {
            const container = document.getElementById('brainCanvasContainer');
            if(!container || brainRenderer) return;

            const width = container.offsetWidth || 600;
            const height = container.offsetHeight || 500;

            brainScene = new THREE.Scene();
            brainScene.background = new THREE.Color(0x090d16);

            brainCamera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
            brainCamera.position.z = 5;

            brainRenderer = new THREE.WebGLRenderer({ antialias: true });
            brainRenderer.setSize(width, height);
            container.appendChild(brainRenderer.domElement);

            const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
            brainScene.add(ambientLight);

            // Outer Translucent Cortex Shell
            const shellGeo = new THREE.SphereGeometry(1.4, 32, 32);
            const shellMat = new THREE.MeshPhongMaterial({ color: 0xa855f7, wireframe: true, transparent: true, opacity: 0.6 });
            brainMesh = new THREE.Mesh(shellGeo, shellMat);
            brainScene.add(brainMesh);

            // Neural Particles
            const count = 300;
            const pGeo = new THREE.BufferGeometry();
            const positions = new Float32Array(count * 3);
            for(let i=0; i<count*3; i+=3) {
                positions[i] = (Math.random() - 0.5) * 2.6;
                positions[i+1] = (Math.random() - 0.5) * 2.6;
                positions[i+2] = (Math.random() - 0.5) * 2.6;
            }
            pGeo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
            const pMat = new THREE.PointsMaterial({ color: 0x38bdf8, size: 0.05 });
            brainParticles = new THREE.Points(pGeo, pMat);
            brainScene.add(brainParticles);

            function animateBrain() {
                requestAnimationFrame(animateBrain);
                if(brainMesh) brainMesh.rotation.y += 0.004;
                if(brainParticles) brainParticles.rotation.y -= 0.002;
                brainRenderer.render(brainScene, brainCamera);
            }
            animateBrain();
        }

           ========================================================================== */
        let opTimerId, seconds = 6135;

        function updateOpTimerDisplay() {
            const hrs = String(Math.floor(seconds / 3600)).padStart(2, '0');
            const mins = String(Math.floor((seconds % 3600) / 60)).padStart(2, '0');
            const secs = String(seconds % 60).padStart(2, '0');
            document.getElementById('opTimerDisplay').innerText = `${hrs}:${mins}:${secs}`;
        }

        function startOpTimer() {
            if(!opTimerId) {
                opTimerId = setInterval(() => {
                    seconds++;
                    updateOpTimerDisplay();
                }, 1000);
            }
        }

        function pauseOpTimer() {
            clearInterval(opTimerId);
            opTimerId = null;
        }

        function resetOpTimer() {
            pauseOpTimer();
            seconds = 0;
            updateOpTimerDisplay();
        }
